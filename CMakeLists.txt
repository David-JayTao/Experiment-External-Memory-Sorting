cmake_minimum_required(VERSION 3.16)
project(ExternalMemorySorting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(external_sort main.cpp src/run_file.cpp src/buffer.cpp
  src/merge_plan.cpp src/projects.cpp src/loser_tree.cpp src/replacement.cpp
  src/kway_merge.cpp tests/test_sort.cpp)
target_include_directories(external_sort PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(external_sort PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(external_sort PRIVATE /W4 /utf-8)
else()
  target_compile_options(external_sort PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME correctness COMMAND external_sort test)
add_test(NAME cli_help COMMAND external_sort --help)
add_test(NAME reject_zero_block COMMAND external_sort project3 --block 0)
add_test(NAME reject_negative_memory COMMAND external_sort project2 --memory -1)
add_test(NAME reject_bad_k COMMAND external_sort project3 --k 1)
add_test(NAME reject_missing_value COMMAND external_sort project1 --size)
set_tests_properties(reject_zero_block reject_negative_memory reject_bad_k reject_missing_value
  PROPERTIES WILL_FAIL TRUE)
