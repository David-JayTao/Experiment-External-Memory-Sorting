cmake_minimum_required(VERSION 3.16)
project(ExternalMemorySorting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(external_sort main.cpp src/run_file.cpp src/buffer.cpp
  src/merge_plan.cpp src/projects.cpp tests/test_sort.cpp)
target_include_directories(external_sort PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(external_sort PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(external_sort PRIVATE /W4 /utf-8)
else()
  target_compile_options(external_sort PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME correctness COMMAND external_sort test)
